--전체 평균가보다 비싼 책: title, price

select title,price from books
where price >(select avg(price) from books)

--좋아요를 한 번이라도 받은 책의 title

select title from books as b
where id in (select book_id from likes)

--각 책 옆에 전체 평균가를 함께: title, price, 전체평균

select title,price,(select avg(price) from books) as 전체평균 from books as b


--좋아요가 하나도 없는 책의 title을 NOT IN 서브쿼리로

select title from books as b
where id not in (select book_id from likes where book_id is not null)

--자기 장르 평균가보다 비싼 책: title, genre, price
<!-- where b.genre=b2.genre 바깥 books와 연결이 핵심 -->
select b.title,b.genre,b.price from books as b
where b.price > (select avg(b2.price) from books as b2 where b.genre=b2.genre)
