--비교+AND — genre가 'IT'이고 price가 30000 이상: title, price (①처럼, 단 >=)

select b.title,b.genre,b.price from books as b
where b.genre = 'IT' and b.price>=30000

--IN — genre가 소설 또는 에세이: title, genre (②)

select b.title,b.genre from books as b
where b.genre in ('소설','에세이')

--BETWEEN — price 10000~20000: title, price (③, 양끝 포함 주의)

select b.title,b.price from books as b
where b.price between 10000 and 20000
order by b.price desc

--LIKE — title에 '코드' 포함: title (④)

select b.title from books as b
where b.title like '%코드%'

--다중 정렬 — genre 오름차순, 같은 genre면 price 내림차순: title, genre, price (⑤)

select b.title, b.genre, b.price, rank() over (partition by b.genre order by b.price desc) as 순위 from books as b
order by b.genre asc,b.price desc

--DISTINCT — genre 목록 (⑥)

select distinct genre from books
order by genre asc 여기서 정렬하면 문자기준으로 정렬하는데 영어먼저 다음 한글

--전체 통계 — 모든 책의 평균가·최고가·최저가·권수 한 행에 (⑦)

select avg(price) as 평균가, max(price) as 최고가, min(price) as 최저가,count(id) as 총권수 from books

--LIMIT — view_count 높은 순 상위 3권: title, view_count (⑧)

select title,view_count from books
order by view_count desc
limit 3

--장르별 책 개수: genre, 개수

select genre,count(genre) from books
group by genre
order by count(genre) desc

--장르별 평균가: genre, 평균가

select genre,avg(price) from books
group by genre
order by genre,avg(price)

--장르별 최고가: genre, 최고가

select genre,max(price) from books
group by genre

--price가 15000 이상인 책만 대상으로, 장르별 개수: genre, 개수

select b.genre,count(b.genre) as 장르개수 from books as b
where b.price>=15000
group by b.genre

--책이 2권 이상인 장르만, 그 장르별 평균가: genre, 평균가

select b.genre, count(b.genre) as 총권수,avg(b.price) as 평균가 from books as b
group by b.genre
having count(b.genre)>=2
