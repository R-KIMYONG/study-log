--[기초 1] likes를 집계해 책별 좋아요 수를 구하는 CTE like_cnt를 만들고, 좋아요가 2개 이상인 책의 제목을 출력해라. (WITH 사용, 서브쿼리 ❌)

with like_cnt as (
	select l.book_id,count(*) as like_count from likes as l
	group by l.book_id
  	
)

select b.title,coalesce(lc.like_count,0) from books as b
left join like_cnt as lc on lc.book_id=b.id
where lc.like_count>=2


--[기초 2] genre별 평균 가격을 구하는 CTE genre_avg를 만들고, 그 평균이 전체 장르 평균가보다 높은 장르만 출력해라. (힌트: genre_avg를 만든 뒤 바깥에서 AVG(avg_price) 같은 전체 평균과 비교)

with genre_avg as (
	select b.genre,avg(b.price) as avg_price from books as b
  	group by b.genre
)

select genre,avg_price from genre_avg
where avg_price > (select avg(avg_price) from genre_avg)


--[기초 3] CTE 2개를 체이닝해서 "책별 좋아요 수 → 좋아요 수 기준 상위 3권"을 뽑아라. (like_cnt → ranked 구조, RANK() OVER (ORDER BY ... DESC) 사용. 좋아요 0인 책도 나오면 좋지만, 일단 좋아요 있는 책 기준으로 시작해도 OK)

with like_cnt as (
	select b.id,b.title,count(l.book_id) as total_likes from books as b
  	left join likes as l on l.book_id = b.id
  	group by b.id
),ranked as (
	select title,total_likes,row_number() over (order by total_likes desc) as rnk from like_cnt
)

select * from ranked
limit 3

--[복습·anti-join] (Day2 복습) 좋아요가 한 번도 없는 책의 제목을 출력해라. NOT EXISTS 또는 LEFT JOIN likes ... WHERE likes.book_id IS NULL 아무거나.


select b.id,b.title from books as b
where not exists (select 1 from likes as l where l.book_id=b.id )

--[상] 재귀 CTE로 1~10 숫자를 생성하고, 그중 짝수만 출력해라. (앵커 + UNION ALL + 재귀 + 종료조건)

with recursive nums as (
	select 1 as n
  	union all
  	select n+1 from nums
  	where n<10
)

select n from nums
where n %2 =0

-- [조립 연습] 목표표: genre, title, like_total, comment_total, rnk_in_genre
-- 책별 좋아요수·댓글수 + 장르별 좋아요 순위. 0인 책도 포함(LEFT JOIN).
-- ============================================================

-- (A) 체이닝: l → lc (앞 CTE를 뒤 CTE가 참조)
-- ⚠️ lc에서 재집계 시 l의 컬럼 전부 GROUP BY에 나열해야 함 (CTE엔 PK 함수종속 안 먹힘)
with l as (
	select b.id, b.title, b.genre, count(likes.book_id) as like_total
	from books as b
	left join likes on likes.book_id = b.id
	group by b.id
), lc as (
	select l.*, count(comments.book_id) as comment_total
	from l
	left join comments on comments.book_id = l.id
	group by l.id, l.title, l.genre, l.like_total
)
select *, rank() over (partition by genre order by like_total desc)
from lc;

-- (B) 병렬: l, c 각각 독립 집계 → 끝에서 books에 LEFT JOIN
-- 독립된 집계는 병렬이 정석 (GROUP BY 간결, books 시작이라 NULL 대신 0)
with l as (
	select b.id, count(likes.book_id) as like_total
	from books as b
	left join likes on likes.book_id = b.id
	group by b.id
), c as (
	select b.id, count(comments.book_id) as comment_cnt
	from books as b
	left join comments on comments.book_id = b.id
	group by b.id
)
select b.genre, b.title, l.like_total, c.comment_cnt,
       rank() over (partition by b.genre order by like_total desc) as rnk_in_genre
from books as b
left join l on l.id = b.id
left join c on c.id = b.id;

-- ============================================================
-- [병렬 연습 2] base = 유저. 유저별 좋아요/댓글 수 + 좋아요 순위
-- ============================================================
with lg as (
	select u.id, count(likes.user_id) as like_given
	from users as u
	left join likes on likes.user_id = u.id
	group by u.id
), cw as (
	select u.id, count(comments.user_id) as comment_written
	from users as u
	left join comments on comments.user_id = u.id
	group by u.id
)
select u.id, u.name, lg.like_given, cw.comment_written,
       rank() over (order by lg.like_given desc) as rnk
from users as u
left join lg on lg.id = u.id
left join cw on cw.id = u.id
order by rnk, u.name;
-- RANK 동점: 좋아요 1개인 최D·정E는 공동 4등.
-- rank() over(order by like_given desc, name) → 동점 '제거'(4,5). 표시만 정렬하려면 바깥 ORDER BY.
