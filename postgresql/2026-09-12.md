--Q1 (INNER) — 댓글과 작성자 이름을 함께: content, name (comments ⨝ users)

select u.name,c.content from users as u
inner join comments as c on c.user_id=u.id

--Q2 (LEFT + count) — 모든 책의 좋아요 수 (좋아요 0인 책도 0으로 나와야 함): title, like_count

select b.title,count(l.book_id) as 총좋아요 from books as b
left join likes as l on l.book_id = b.id
group by b.title

--Q3 (LEFT + IS NULL) — 댓글이 하나도 없는 책의 title (안티조인)

select b.title from books as b
left join comments as c on c.book_id = b.id
group by b.title,c.content
having c.content is null

--Q4 — genre별로 price 비싼 순 순위: title, genre, price, 순위 (rank() over (...))

select b.title,b.genre,b.price, rank() over (partition by b.genre order by b.price desc) from books as b

--Q5 — 좋아요를 2개 이상 받은 책의 title, 좋아요수, 좋아요 많은 순

select b.title,count(l.book_id) as 좋아요수 from books as b
left join likes as l on l.book_id = b.id
group by b.title
having count(l.book_id)>=2
order by count(l.book_id) desc
