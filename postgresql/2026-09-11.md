1. price가 20000 이상인 책의 title, price를 price 내림차순으로.
   select title from books
   where price >=20000
   order by price desc
2. genre별 책 개수를 세고, 개수 많은 순으로.
   select genre,count(\*) as cnt from books
   group by genre
   order by cnt desc
3. genre별 평균 price를 구하되, 평균이 15000 이상인 genre만 (반올림 없이 OK).
   select genre,avg(price) as p from books
   group by genre
   Having avg(price)>=15000
4. 좋아요가 하나도 없는 책의 title. (LEFT JOIN 또는 NOT EXISTS)
   select b.id,b.title from books as b
   left join likes l on l.book_id=b.id
   where l.book_id is null
   order by b.id desc
5. 책별 좋아요 수를 구해 top 5 (좋아요 많은 순, title + 좋아요수).
   select b.id,b.title,count(l.user_id) as like_count from books as b
   left join likes l on l.book_id=b.id
   group by b.id,b.title
   order by like_count desc
   limit 5
6. 댓글을 한 번이라도 단 유저의 이름 (중복 없이).
   select u.id,u.name from users as u
   left join comments as c on c.user_id = u.id
   where c.user_id is not null
   group by u.id,u.name
7. 각 genre에서 가장 비싼 책의 title, genre, price.
   select DISTINCT ON (b.genre) b.title, b.genre,b.price from books as b
   order by genre, price desc
8. CTE 사용 — genre별 평균가를 먼저 구한 뒤, 자기 genre 평균보다 비싼 책들의 title, genre, price.
   with GenreAvg as (
   SELECT genre, AVG(price) AS avg_price from books
   group by genre
   )

select b.title,b.genre,b.price,gen.avg_price from books as b
inner join GenreAvg as gen on b.genre = gen.genre
where b.price>gen.avg_price 9. 윈도우 함수 — genre별로 price가 비싼 순 순위를 매겨 title, genre, price, 순위. (rank() over (...))
select title,genre,price,rank() over (partition by genre order by price desc) as rank

with numbered_comments as (

select u.id,u.name,c.content,c.created_at,row_number() over (partition by name order by c.created_at desc) as rowct from users as u
left join comments as c on c.user_id = u.id
where c.created_at is not null
)

select id, name, created_at ,rowct,content from numbered_comments
where rowct=1
